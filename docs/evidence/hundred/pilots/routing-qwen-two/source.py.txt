import asyncio, copy, json, sys, time
from pathlib import Path
from types import SimpleNamespace
import httpx
REPO=Path('/workspace/scratch/8a803cd8931f/openclaw-long-tasks')
sys.path.insert(0,str(REPO/'scripts'))
from pilot_hanoi import local_endpoint
from long_tasks.hanoi_choices import ChoiceHanoi, CHOICE_FORMAT
from long_tasks.hanoi import oracle_moves, apply_move

async def run(args, endpoint):
 task=ChoiceHanoi(7)
 state=task.initial_state
 states=[]
 for expected in oracle_moves(7):
  if state['step'] in [0,2,8,63]: states.append((state,expected))
  state={'pegs':apply_move(state['pegs'],expected),'step':state['step']+1}
 plans=[]
 for style in ['routes','names','predicate']:
  for state,expected in states:
   for ix,seed in enumerate([503,509,521]):
    opts=task.options(state)
    names=['red','blue','green'] if style=='names' else ['0','1','2']
    if state['step']%2==0:
     rule=f"Move DISK 1 one arrow forward in this directed cycle: {names[0]} -> {names[2]} -> {names[1]} -> {names[0]}. Do not go backwards."
     if style=='predicate':
      rule="Select the row satisfying this Python boolean expression: disk == 1 and destination == {0: 2, 1: 0, 2: 1}[source]. Evaluate the dictionary lookup before comparing destination."
    else:
     rule="Select the row with the largest DISK number."
    rows=[f"{k}: DISK {m[0]}, ROUTE {names[m[1]]} -> {names[m[2]]}" for k,m in opts.items()]
    if style=='predicate': rows=[f"{k}: disk={m[0]}, source={m[1]}, destination={m[2]}" for k,m in opts.items()]
    prompt=rule+'\nOPTIONS:\n'+'\n'.join(rows)+'\nReturn only JSON with the selected choice letter: {"choice":"A"}, {"choice":"B"}, or {"choice":"C"}. /no_think'
    plans.append({'style':style,'state':state,'expected':expected,'seed':seed,'prompt':prompt,'format':CHOICE_FORMAT})
 (args.output/'protocol.json').write_text(json.dumps({'closed_loop':False,'description':'adaptive route representation pilot after fixed and paired rules failed','plans':plans},indent=2))
 rows=[]
 async with httpx.AsyncClient(timeout=180,trust_env=False) as c:
  for plan in plans:
   start=time.monotonic()
   r=await c.post(endpoint+'/chat/completions',json={'model':'local-single','messages':[{'role':'user','content':plan['prompt']}],'max_tokens':192 if plan['style']=='reasoned' else 32,'temperature':0.1,'seed':plan['seed'],'response_format':plan['format']})
   r.raise_for_status(); value=r.json(); choice=value['choices'][0]; text=choice['message']['content']
   try: correct=task.parse(json.dumps({'choice':json.loads(text)['choice']}),plan['state']).action==plan['expected'] and choice['finish_reason']=='stop'
   except Exception: correct=False
   row={**plan,'response':value,'correct':correct,'elapsed_seconds':round(time.monotonic()-start,3)}; rows.append(row)
   (args.output/'samples.json').write_text(json.dumps(rows,indent=2))
   print(json.dumps({'style':plan['style'],'step':plan['state']['step'],'correct':correct,'answer':text}),flush=True)
 (args.output/'results.json').write_text(json.dumps({style:{'correct':sum(r['correct'] for r in rows if r['style']==style),'calls':sum(r['style']==style for r in rows)} for style in ['routes','names','predicate']},indent=2))
args=SimpleNamespace(llama=Path('/tmp/maker-runtime/llama-b10867/llama-server'),model_file=Path('/tmp/codex-model-cache/qwen2.5-3b-instruct-q4_k_m.gguf'),swarm=REPO.parent/'swarm-of-experts',swarm_python=Path('/tmp/swarm-clean/bin/python'),output=Path('/tmp/maker-hundred-pilots/routing-qwen-two'))
with local_endpoint(args) as endpoint: asyncio.run(run(args,endpoint))
